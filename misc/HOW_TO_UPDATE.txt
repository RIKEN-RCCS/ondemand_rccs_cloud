# アップデート
## 不要なファイルを消す
```
for i in bc_desktop projects system-status module-browser myjobs; do
  rm -rf ${i}.bak
  mv ${i} ${i}.bak
done
```

## シンボリックリンクを作成する
ln -s /var/www/ood/apps/sys/ondemand_rccs_cloud/misc/global_bc_items.yml /etc/ood/config/ondemand.d/global_bc_items.yml
ln -s /var/www/ood/apps/sys/ondemand_rccs_cloud/misc/missing_home_directory.html /etc/ood/config/pun/html/missing_home_directory.html
ln -s /var/www/ood/apps/sys/ondemand_rccs_cloud/misc/ondemand.yml /etc/ood/config/ondemand.d/

# 初期インストール
## リポジトリのクローン
```
cd /var/www/ood/apps/sys
git clone git@github.com:RIKEN-RCCS/ondemand_rccs_cloud.git
for i in Desktop Gnuplot GrADS ImageJ JupyterLab OVITO ParaView PyMOL Smokeview SshPublicKey ttyd VisIt VSCode WHEEL Slurm
do
  ln -s ondemand_rccs_cloud/$i .
done
mkdir -p /etc/ood/config/pun/html
mkdir -p /etc/ood/config/apps/dashboard/initializers
mkdir -p /etc/ood/config/apps/shell
git clone git@github.com:OpenOnDemandJP/OpenComposer.git
cd OpenComposer/
mv manifest.yml manifest.yml.bak
cp conf.yml.erb.sample conf.yml.erb
git clone git@github.com:RIKEN-RCCS/composer_rccs_cloud.git
```

conf.yml.erbに下記を追加
```
apps_dir: "./composer_rccs_cloud/"
login_node: "login2.cloud.r-ccs.riken.jp"
```
